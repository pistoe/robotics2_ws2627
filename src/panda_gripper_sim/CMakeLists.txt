cmake_minimum_required(VERSION 3.16)
project(panda_gripper_sim)
find_package(ament_cmake REQUIRED)
find_package(mujoco_ros2_control_plugins REQUIRED)
find_package(mujoco_vendor REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(franka_msgs REQUIRED)
find_package(control_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
add_library(panda_gripper_sim SHARED src/gripper.cpp)
target_compile_features(panda_gripper_sim PUBLIC cxx_std_17)
target_compile_options(panda_gripper_sim PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(panda_gripper_sim mujoco_ros2_control_plugins pluginlib rclcpp_action
  franka_msgs control_msgs sensor_msgs std_srvs)
target_link_libraries(panda_gripper_sim mujoco::mujoco)
pluginlib_export_plugin_description_file(mujoco_ros2_control_plugins plugin.xml)
install(TARGETS panda_gripper_sim LIBRARY DESTINATION lib)
ament_package()
