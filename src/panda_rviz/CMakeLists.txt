cmake_minimum_required(VERSION 3.16)
project(panda_rviz)
find_package(ament_cmake REQUIRED)
find_package(pluginlib REQUIRED)
find_package(message_filters REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_default_plugins REQUIRED)
add_library(panda_rviz_image SHARED src/image_display.cpp)
target_compile_features(panda_rviz_image PRIVATE cxx_std_17)
target_compile_options(panda_rviz_image PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(panda_rviz_image rviz_default_plugins::rviz_default_plugins pluginlib::pluginlib)
ament_target_dependencies(panda_rviz_image message_filters)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS panda_rviz_image DESTINATION lib)
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
